def _btn(text: str, callback_data: str = None, url: str = None,
         icon_id: str = None, style: str = "primary") -> dict:
    btn: dict = {"text": text}
    if callback_data:
        btn["callback_data"] = callback_data
    if url:
        btn["url"] = url
    btn["icon_custom_emoji_id"] = icon_id or ""
    btn["style"] = style
    return btn

# ╔══════════════════════════════════════════════════════════════════════════╗
# ║                    RESULTS ZIP SENDERS                                  ║
# ╚══════════════════════════════════════════════════════════════════════════╝

async def _send_zip(chat_id, uid, bot: Bot):
    r    = user_results[uid]
    mode = r["mode"]

    if mode == "surfshark":
        premium = r["ss_premium"]; free = r["ss_free"]; dead = r["ss_dead"]
        total   = r["ss_total"]
        if not total:
            await bot.send_message(chat_id, f"{E_EMPTY} No Surfshark results yet.", parse_mode="HTML", reply_markup=main_markup(mode))
            return
        revived = sum(1 for x in premium if x.get("revived"))
        buf = io.BytesIO()
        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
            for res in premium:
                plan_slug  = re.sub(r"[^\w]", "_", res.get("plan", "Unknown"))
                email_slug = re.sub(r"[^\w\-_]", "_", res.get("email", "unk").split("@")[0])
                zf.writestr(f"premium/{plan_slug}/{email_slug}_{res.get('expires_at','unk')}.txt", res.get("content", ""))
            for res in free:
                email_slug = re.sub(r"[^\w\-_]", "_", res.get("email", "unk").split("@")[0])
                zf.writestr(f"free/free_{email_slug}.txt", res.get("content", ""))
            lines = [
                "Surfshark Cookie Check Summary", "=" * 50,
                f"Total   : {total}", f"Premium : {len(premium)}", f"Free    : {len(free)}",
                f"Dead    : {len(dead)}", f"Revived : {revived}", "", "── PREMIUM DETAILS ──",
            ]
            for res in premium:
                lines.append(
                    f"  {res.get('email','?'):<32} | {res.get('plan','?'):<28} "
                    f"| {res.get('days_left','?')} days | exp {res.get('expires_at','?')}"
                )
            lines.append(f"\nGenerated: {datetime.now():%Y-%m-%d %H:%M:%S}")
            zf.writestr("summary.txt", "\n".join(lines))
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        cap = (f"{E_ZIP} <b>Surfshark Results — {ts}</b>\n\n"
               f"{SS_PREMIUM} Premium : <b>{len(premium)}</b>\n"
               f"{SS_FREE} Free    : <b>{len(free)}</b>\n"
               f"{SS_DEAD} Dead    : <b>{len(dead)}</b>\n"
               f"{E_COUNT} Total   : <b>{total}</b>")
        await bot.send_document(chat_id,
            document=BufferedInputFile(buf.getvalue(), filename=f"surfshark_results_{ts}.zip"),
            caption=cap, parse_mode="HTML", reply_markup=main_markup(mode))

    elif mode == "spotify":
        premium = r["sp_premium"]; free = r["sp_free"]; dead = r["sp_dead"]
        total   = r["sp_total"]
        if not total:
            await bot.send_message(chat_id, f"{E_EMPTY} No Spotify results yet.", parse_mode="HTML", reply_markup=main_markup(mode))
            return
        tmp = tempfile.mktemp(suffix=".zip")
        try:
            with zipfile.ZipFile(tmp, "w", zipfile.ZIP_DEFLATED) as zf:
                for res in premium + free + dead:
                    folder = {"PREMIUM": "premium", "FREE": "free", "DEAD": "dead"}.get(res["status"], "dead")
                    name   = re.sub(r"[^\w\-.]", "_", res.get("filename", "cookie")) or "cookie.txt"
                    zf.writestr(f"{folder}/{name}", res.get("content", ""))
            ts = int(time.time())
            await bot.send_document(
                chat_id,
                document=FSInputFile(tmp, filename=f"spotify_results_{ts}.zip"),
                caption=(f"{E_ZIP}  <b>Spotify Results</b>\n"
                         f"{SP_PREMIUM} Premium: <b>{len(premium)}</b>  "
                         f"{SP_FREE} Free: <b>{len(free)}</b>  "
                         f"{SP_DEAD} Dead: <b>{len(dead)}</b>"),
                parse_mode="HTML", reply_markup=main_markup(mode),
            )
        finally:
            try:
                os.remove(tmp)
            except Exception:
                pass

    elif mode == "netflix":
        premium = r["nf_premium"]; free_l = r["nf_free"]
        dead    = r["nf_dead"];    tokens = r["nf_tokens"]
        total   = r["nf_total"];   sub    = r.get("nf_sub", "check")
        if not total:
            await bot.send_message(chat_id, f"{E_EMPTY} No Netflix results yet.", parse_mode="HTML", reply_markup=main_markup(mode))
            return
        if sub == "nftoken":
            buf = io.BytesIO()
            with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
                for idx, res in enumerate(tokens, 1):
                    zf.writestr(f"tokens/token_{idx}.txt", nf_build_token_export(res, idx))
                zf.writestr("summary.txt", f"Netflix Token Gen\nTotal: {total}\nTokens: {len(tokens)}\nFailed: {len(dead)}\nGenerated: {datetime.now():%Y-%m-%d %H:%M:%S}")
            ts  = datetime.now().strftime("%Y%m%d_%H%M%S")
            cap = (f"{E_ZIP} <b>Netflix Tokens — {ts}</b>\n\n"
                   f"{SP_CODE} Tokens : <b>{len(tokens)}</b>\n"
                   f"{SP_DEAD} Failed : <b>{len(dead)}</b>\n"
                   f"{E_COUNT} Total  : <b>{total}</b>")

async def _gc_store_sp(bot: Bot, premium: list):
    if not GC_STORE_CHAT_ID or not premium:
        return
    tmp = tempfile.mktemp(suffix=".zip")
    try:
        with zipfile.ZipFile(tmp, "w", zipfile.ZIP_DEFLATED) as zf:
            for r in premium:
                name = re.sub(r"[^\w\-.]", "_", r.get("filename", "cookie")) or "cookie.txt"
                zf.writestr(name, r.get("content", ""))
        ts = int(time.time())
        await bot.send_document(
            GC_STORE_CHAT_ID,
            document=FSInputFile(tmp, filename=f"sp_gc_{ts}.zip"),
            caption=f"{E_ZIP} <b>Spotify GC</b>\n{SP_PREMIUM} Premium: <b>{len(premium)}</b>",
            parse_mode="HTML", disable_notification=True,
        )
    except Exception as e:
        logger.warning(f"GC sp error: {e}")
    finally:
        try:
            os.remove(tmp)
        except Exception:
            pass

@router.message(F.photo)
async def handle_photo(message: types.Message, bot: Bot):
    uid     = message.from_user.id
    chat_id = message.chat.id
    r       = user_results[uid]

    if not r.get("hs_awaiting_qr"):
        return

    user_results[uid]["hs_awaiting_qr"] = False
    scanning_msg = await message.answer("🔍 <b>Scanning QR code…</b>", parse_mode="HTML")

    try:
        photo_file  = await message.photo[-1].get_file()
        buf         = io.BytesIO()
        await bot.download_file(photo_file.file_path, destination=buf)
        image_bytes = buf.getvalue()

        loop    = asyncio.get_event_loop()
        qr_data = await loop.run_in_executor(None, scan_qr, image_bytes)

        if not qr_data:
            await safe_edit(scanning_msg,
                "❌ <b>No QR code found in image.</b>\nSend a clearer photo of the TV screen.",
                markup=main_markup("hotstar"))
            hs_vault_record(False)
            return

        if not any(d in qr_data for d in ("hotstar.com", "jiohotstar.com")):
            await safe_edit(scanning_msg,
                "❌ <b>Not a JioHotstar QR code.</b>\n\n"
                f"Decoded: <code>{qr_data[:100]}</code>\n\n"
                "Only QR codes from hotstar.com / jiohotstar.com are supported.",
                markup=main_markup("hotstar"))
            hs_vault_record(False)
            return

        await safe_edit(scanning_msg, f"{E_GEAR} <b>QR decoded. Picking vault account…</b>")

        cookie_info, err = hs_vault_pick_cookie()
        if not cookie_info:
            await safe_edit(scanning_msg,
                f"❌ <b>No vault accounts available.</b>\n\n{err}\n\n"
                f"Ask an admin to upload more via <b>{E_UPLOAD} Upload Vault</b>.",
                markup=main_markup("hotstar"))
            hs_vault_record(False)
            return

        use_count = hs_vault_get_use_count(cookie_info["filename"])
        await safe_edit(scanning_msg,
            f"{E_CHECK} <b>Account selected</b>  (usage: {use_count}/{HS_VAULT_MAX_USES})\n\n"

@router.message(F.document)
async def handle_file(message: types.Message, bot: Bot):
    uid     = message.from_user.id
    chat_id = message.chat.id
    doc     = message.document
    r       = user_results[uid]
    mode    = r["mode"]

    if not doc:
        await message.answer("Please send a file.")
        return
    if r["processing"]:
        await message.answer(f"{E_CLOCK} Already processing. Please wait.", parse_mode="HTML")
        return

    fname = doc.file_name or ""
    ext   = fname.rsplit(".", 1)[-1].lower() if "." in fname else ""

    if r.get("nf_vault_upload") and ext == "zip":
        user_results[uid]["nf_vault_upload"] = False
        status_msg = await message.answer(f"{E_DOWN} <b>Downloading ZIP…</b>", parse_mode="HTML")
        try:
            tg_file  = await bot.get_file(doc.file_id)
            zip_path = os.path.join(tempfile.gettempdir(), f"nf_vault_{uid}_{int(time.time())}.zip")
            await bot.download_file(tg_file.file_path, destination=zip_path)
            await safe_edit(status_msg, f"{E_FOLDER} <b>Extracting cookies into Netflix vault…</b>")
            added, err = nf_vault_add_zip(zip_path)
            try:
                os.remove(zip_path)
            except Exception:
                pass
            if err and added == 0:
                await safe_edit(status_msg,
                    f"❌ <b>Upload failed:</b> <code>{err}</code>",
                    markup=main_markup("netflix"))
            else:
                await safe_edit(status_msg,
                    f"✅ <b>Netflix Vault Updated!</b>\n\n"
                    f"📦 Added  : <b>{added}</b> cookie file(s)\n"
                    f"📂 Total  : <b>{nf_vault_count()}</b>",
                    markup=main_markup("netflix"))
        except Exception as e:
            await safe_edit(status_msg,
                f"<b>Error during upload:</b> <code>{str(e)[:200]}</code>",
                markup=main_markup("netflix"))
        return

    if r.get("hs_vault_upload") and ext == "zip":
        user_results[uid]["hs_vault_upload"] = False
        status_msg = await message.answer(f"{E_DOWN} <b>Downloading ZIP…</b>", parse_mode="HTML")
        try:
            tg_file  = await bot.get_file(doc.file_id)
            zip_path = os.path.join(tempfile.gettempdir(), f"hs_vault_{uid}_{int(time.time())}.zip")
            await bot.download_file(tg_file.file_path, destination=zip_path)
            await safe_edit(status_msg, f"{E_FOLDER} <b>Extracting cookies into vault…</b>")
            added, err = hs_vault_add_zip(zip_path)
            try:
                os.remove(zip_path)
            except Exception:
                pass
            if err and added == 0:
                await safe_edit(status_msg,
                    f"❌ <b>Upload failed:</b> <code>{err}</code>",
                    markup=main_markup("hotstar"))
            else:
                await safe_edit(status_msg,
                    f"✅ <b>Vault updated!</b>\n\n"
                    f"📦 Added : <b>{added}</b> cookie file(s)\n"
                    f"📂 Total  : <b>{hs_vault_count()}</b>\n"
                    f"🟢 Available: <b>{hs_vault_available_count()}</b>",
                    markup=main_markup("hotstar"))
        except Exception as e:
            await safe_edit(status_msg,
                f"<b>Error during upload:</b> <code>{str(e)[:200]}</code>",
                markup=main_markup("hotstar"))
        return

    if ext not in ("txt", "json", "zip"):
        await message.answer(
            f"Unsupported format. Send <code>.txt</code>, <code>.json</code>, or <code>.zip</code>.",
            parse_mode="HTML",
        )
        return
    if doc.file_size > MAX_FILE_SIZE:
        await message.answer(f"File too large. Max: {MAX_FILE_SIZE // 1024 // 1024} MB.", parse_mode="HTML")
        return

    user_results[uid]["processing"] = True
    status_msg = await message.answer(f"{E_DOWN} <b>Downloading file…</b>", parse_mode="HTML")
    temp_path  = None

    try:
        tg_file   = await bot.get_file(doc.file_id)
        temp_path = os.path.join(tempfile.gettempdir(), f"{mode}_{uid}_{int(time.time())}_{fname}")
        await bot.download_file(tg_file.file_path, destination=temp_path)

        if ext == "zip":
            await safe_edit(status_msg, f"{E_FOLDER} <b>Extracting ZIP…</b>")
            files_to_check = extract_zip_cookies(temp_path, prefix=mode)
            if not files_to_check:
                await safe_edit(status_msg, "No valid cookie files found in ZIP.")
                user_results[uid]["processing"] = False
                return
        else:
            files_to_check = [temp_path]

        total = len(files_to_check)
        await safe_edit(status_msg,
            f"{E_GEAR} <b>Checking {total} cookie(s) via {SERVICES[mode]}…</b>\n"
            f"Using <b>{MAX_WORKERS}</b> parallel workers.",
        )

        if mode == "netflix" and r.get("nf_sub") == "clean":
            await safe_edit(status_msg, f"{SP_SPARK}  <b>Cleaning &amp; deduplicating cookies…</b>")
            try:
                raw_text = open(temp_path, "r", encoding="utf-8", errors="ignore").read() \
                           if ext != "zip" else \
                           "\n".join(open(fp, "r", encoding="utf-8", errors="ignore").read()
                                     for fp in files_to_check)
                zip_buf, valid = nf_clean_cookies_to_zip(raw_text)
                if not zip_buf or valid == 0:
                    await safe_edit(status_msg,
                        f"{SP_DEAD} <b>No valid Netflix cookies found.</b>\nCheck your file format.",
                        markup=main_markup("netflix"))
                    user_results[uid]["processing"] = False
                    return
                ts = datetime.now().strftime("%Y%m%d_%H%M%S")
                await safe_edit(status_msg,
                    f"{SP_SPARK}  <b>Done!</b>  Found <b>{valid}</b> unique cookie(s).")
                await bot.send_document(
                    chat_id,
                    document=BufferedInputFile(zip_buf.getvalue(), filename=f"netflix_clean_{ts}.zip"),
                    caption=(f"{SP_PREMIUM} <b>Netflix Clean Cookies</b>\n\n"
                             f"{E_COUNT} Valid / unique: <b>{valid}</b>\n"
                             f"Format: Netscape .txt (one file per cookie)"),
                    parse_mode="HTML", reply_markup=main_markup("netflix"))
            except Exception as e:
                await safe_edit(status_msg, f"<b>Clean error:</b> <code>{str(e)[:200]}</code>")
            user_results[uid]["processing"] = False
            return

        if mode == "surfshark":
            await _run_surfshark(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "spotify":
            await _run_spotify(uid, chat_id, total, files_to_check, ext, temp_path, status_msg, bot)
        elif mode == "hotstar":
            await _run_hotstar(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "hbomax":
            await _run_hbomax(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "chatgpt":
            await _run_chatgpt(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "cursor":
            await _run_cursor(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "instagram":
            await _run_instagram(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        elif mode == "scan":
            await _run_scan(uid, chat_id, total, files_to_check, ext, status_msg, bot)
        else:
            await _run_netflix(uid, chat_id, total, files_to_check, ext, status_msg, bot)

        for fp in files_to_check:
            try:
                if fp != temp_path and os.path.exists(fp):
                    os.remove(fp)
            except Exception:
                pass

    except Exception as e:
        logger.exception("handle_file error")
        await safe_edit(status_msg, f"<b>Error:</b> <code>{str(e)[:200]}</code>")
        user_results[uid]["processing"] = False
    finally:
        if temp_path and os.path.exists(temp_path):
            try:
                os.remove(temp_path)
            except Exception:
                pass

async def _run_chatgpt(uid, chat_id, total, files, ext, status_msg, bot: Bot):
    premium = []; free = []; dead = []
    processed = 0; last_edit = 0.0; start_t = time.monotonic()
    loop = asyncio.get_event_loop()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [loop.run_in_executor(executor, cg_check_file, fp) for fp in files]
        for coro in asyncio.as_completed(futures):
            try:
                result = await coro
            except Exception as e:
                result = {"status": "DEAD", "reason": str(e)[:50], "filename": "?", "content": ""}

            s = result.get("status")
            if s == "PREMIUM":
                premium.append(result)
                await bot.send_message(chat_id, cg_fmt_premium(result), parse_mode="HTML")
            elif s == "FREE":
                free.append(result)
            else:
                dead.append(result)

            processed += 1
            if total > 1:
                now = time.monotonic()
                if now - last_edit >= 2.0:
                    last_edit = now
                    await safe_edit(status_msg, cg_fmt_progress(
                        processed, total, len(premium), len(free), len(dead)))

    elapsed = time.monotonic() - start_t
    user_results[uid].update({
        "cg_premium": premium, "cg_free": free, "cg_dead": dead,
        "cg_total": total, "processing": False,
    })

    if ext != "zip" and total == 1:
        only = (premium or free or dead)[0]
        await safe_edit(status_msg, cg_fmt_single(only), markup=main_markup("chatgpt"))
    else:
        await safe_edit(status_msg,
            cg_fmt_summary(len(premium), len(free), len(dead), total, elapsed),
            markup=main_markup("chatgpt"))
        if premium:
            buf = io.BytesIO()
            with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
                for idx, hit in enumerate(premium, 1):
                    email_slug = re.sub(r"[^\w\-_]", "_", hit.get("email", "unk").split("@")[0])
                    body = (
                        f"Name    : {hit.get('name','Unknown')}\n"
                        f"Email   : {hit.get('email','Unknown')}\n"
                        f"Plan    : {hit.get('plan_upper','Unknown')}\n"
                        f"Expires : {hit.get('expiration','Unknown')}\n\n"
                        f"Cookies (Netscape):\n"
                        f"{cg_format_cookie_netscape('__Secure-next-auth.session-token.0', hit.get('cookie_0',''))}\n"
                        f"{cg_format_cookie_netscape('__Secure-next-auth.session-token.1', hit.get('cookie_1',''))}\n"
                    )
                    zf.writestr(f"premium/{email_slug}.txt", body)
            ts = datetime.now().strftime("%Y%m%d_%H%M%S")
            await bot.send_document(
                chat_id,
                document=BufferedInputFile(buf.getvalue(), filename=f"chatgpt_hits_{ts}.zip"),
                caption=f"{CG_PREMIUM} <b>{len(premium)} ChatGPT premium hit(s)</b>",
                parse_mode="HTML", reply_markup=main_markup("chatgpt"),
            )

async def _run_scan(uid, chat_id, total, files, ext, status_msg, bot: Bot):
    live_hits = []; dead_hits = []
    processed = 0; last_edit = 0.0; start_t = time.monotonic()
    loop = asyncio.get_event_loop()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [loop.run_in_executor(executor, scan_check_file, fp) for fp in files]
        for coro in asyncio.as_completed(futures):
            try:
                result = await coro
            except Exception as e:
                result = {"error": str(e)[:60], "filename": "?"}

            if "error" in result:
                dead_hits.append(result)
                processed += 1
                continue

            for service, sres in result.get("results", {}).items():
                if sres.get("status") == "LIVE":
                    hit = {
                        "service":  service,
                        "username": sres.get("username"),
                        "filename": result.get("filename"),
                        "content":  result.get("content", ""),
                    }
                    live_hits.append(hit)

            processed += 1

            if total > 1:
                now = time.monotonic()
                if now - last_edit >= 2.0:
                    last_edit = now
                    await safe_edit(status_msg, scan_fmt_progress(
                        processed, total, len(live_hits), len(dead_hits)))

            if total == 1:
                await safe_edit(status_msg, scan_fmt_result(result),
                                markup=main_markup("scan"))

    elapsed = time.monotonic() - start_t
    user_results[uid].update({
        "scan_live": live_hits, "scan_dead": dead_hits,
        "scan_total": total, "processing": False,
    })

    if total > 1:
        await safe_edit(status_msg,
            f"{E_CHECK} <b>Cookie Scanner Complete!</b>\n\n"
            f"━━━━━━━━━━━━━━━━━━━━━━━\n"
            f"{E_COUNT} Files : <b>{total}</b>\n"
            f"{SCAN_LIVE} Live  : <b>{len(live_hits)}</b>\n"
            f"{SCAN_DEAD} Dead  : <b>{len(dead_hits)}</b>\n"
            f"{E_CLOCK} Time  : <b>{elapsed:.1f}s</b>\n"
            f"━━━━━━━━━━━━━━━━━━━━━━━",
            markup=main_markup("scan"))

    if live_hits:
        buf = io.BytesIO()
        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
            seen_files = set()
            for hit in live_hits:
                srv = hit["service"]
                fname = hit["filename"]
                key = f"{srv}_{fname}"
                if key in seen_files:
                    continue
                seen_files.add(key)
                zf.writestr(f"{srv}/{fname}", hit["content"])
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        await bot.send_document(
            chat_id,
            document=BufferedInputFile(buf.getvalue(), filename=f"scanner_live_{ts}.zip"),
            caption=f"{SCAN_LIVE} <b>{len(live_hits)} live cookie file(s)</b>",
            parse_mode="HTML", reply_markup=main_markup("scan"),
        )

async def _run_surfshark(uid, chat_id, total, files, ext, status_msg, bot: Bot):
    premium_list = []; free_list = []; dead_list = []
    processed = revived_n = 0; last_edit = 0.0
    loop = asyncio.get_event_loop()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [loop.run_in_executor(executor, ss_check_file, fp) for fp in files]
        for coro in asyncio.as_completed(futures):
            try:
                result = await coro
            except Exception as e:
                result = {"status": "DEAD", "reason": str(e)[:50], "filename": "?", "content": ""}

            status = result.get("status")
            if status == "PREMIUM":
                premium_list.append(result)
                if result.get("revived"):
                    revived_n += 1
                slug = re.sub(r"[^\w\-_]", "_", result.get("email", "unk").split("@")[0])
                with open(os.path.join("ss_premium", f"{slug}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(result.get("content", ""))
            elif status == "FREE":
                free_list.append(result)
                slug = re.sub(r"[^\w\-_]", "_", result.get("email", "unk").split("@")[0])
                with open(os.path.join("ss_free", f"free_{slug}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(result.get("content", ""))
            else:
                dead_list.append(result)
            processed += 1
            if total > 1:
                now = time.monotonic()
                if now - last_edit >= 2.0:
                    last_edit = now
                    await safe_edit(status_msg, ss_fmt_progress(processed, total, len(premium_list), len(free_list), len(dead_list), revived_n))

    user_results[uid].update({
        "ss_premium": premium_list, "ss_free": free_list,
        "ss_dead": dead_list, "ss_total": total, "processing": False,
        "ss_awaiting_code": False, "ss_login_data": None,
    })

    if ext != "zip" and total == 1:
        only = (premium_list or free_list or dead_list)[0]
        await safe_edit(status_msg, ss_fmt_single(only), markup=main_markup("surfshark"))
    else:
        await safe_edit(status_msg,
            ss_fmt_summary(total, len(premium_list), len(free_list), len(dead_list), revived_n),
            markup=main_markup("surfshark"),
        )
        if premium_list:
            ts_str  = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            divider = "═" * 50
            lines   = ["Surfshark Premium Hit Report", divider,
                       f"Generated  : {ts_str}", f"Premium    : {len(premium_list)}",
                       f"Revived    : {revived_n}", divider, ""]

                lines += [f"HIT #{i:03d}",
                          f"  Email    : {res.get('email','?')}",
                          f"  Plan     : {res.get('plan','?')}",
                          f"  Expiry   : {res.get('expires_at','?')}",
                          f"  Days     : {res.get('days_left','?')} days",
                          f"  Price    : {res.get('price','?')}",
                          f"  Country  : {res.get('country','?')}",
                          f"  Revived  : {'Yes' if res.get('revived') else 'No'}",

async def _run_spotify(uid, chat_id, total, files, ext, temp_path, status_msg, bot: Bot):
    premium_list = []; free_list = []; dead_list = []
    start_time = time.time(); processed = 0
    UPDATE_EVERY = max(1, total // 20)
    loop = asyncio.get_event_loop()

    for i, fpath in enumerate(files, 1):
        result = await loop.run_in_executor(None, sp_check_file, fpath)
        if fpath != temp_path:
            try:
                os.remove(fpath)
            except Exception:
                pass
        status = result.get("status")
        if status == "PREMIUM":
            premium_list.append(result)
            name = re.sub(r"[^\w\-.]", "_", result.get("filename", "c")) or "c.txt"
            with open(os.path.join("sp_premium", name), "w") as fh:
                fh.write(result.get("content", ""))
        elif status == "FREE":
            free_list.append(result)
            name = re.sub(r"[^\w\-.]", "_", result.get("filename", "c")) or "c.txt"
            with open(os.path.join("sp_free", name), "w") as fh:
                fh.write(result.get("content", ""))
        else:
            dead_list.append(result)
        processed += 1
        user_results[uid]["sp_total"] += 1
        if processed % UPDATE_EVERY == 0 or processed == total:
            await safe_edit(status_msg, sp_fmt_progress(processed, total, len(premium_list), len(free_list), len(dead_list)))

    elapsed = time.time() - start_time
    user_results[uid].update({
        "sp_premium": premium_list, "sp_free": free_list,
        "sp_dead": dead_list, "sp_total": total, "processing": False,
    })

    if ext != "zip" and total == 1:
        only = (premium_list or free_list or dead_list)[0]
        await safe_edit(status_msg, sp_fmt_premium(only) if only["status"] == "PREMIUM" else

                    parse_mode="HTML", reply_markup=main_markup("spotify"),
                )
            finally:
                try:
                    os.remove(tmp)
                except Exception:
                    pass

async def _run_hotstar(uid, chat_id, total, files, ext, status_msg, bot: Bot):
    live_list = []; device_list = []; free_list = []; inv_list = []
    processed = last_edit = 0.0; live_hit_n = dev_hit_n = 0
    loop = asyncio.get_event_loop()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [loop.run_in_executor(executor, hs_check_file, fp) for fp in files]
        for coro in asyncio.as_completed(futures):
            try:
                result = await coro
            except Exception as e:
                result = {"status": "INVALID", "reason": str(e)[:50], "filename": "?"}

            status = result.get("status", "INVALID")
            if status == "LIVE":
                live_hit_n += 1; result["hit_num"] = live_hit_n
                live_list.append(result)
                with open(os.path.join("hs_live", f"live_{result.get('name','unk')}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(hs_convert_to_netscape(result.get("cookies", {})))
            elif status == "DEVICE_LIMIT":
                dev_hit_n += 1; result["hit_num"] = dev_hit_n
                device_list.append(result)
                with open(os.path.join("hs_device", f"device_{result.get('name','unk')}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(hs_convert_to_netscape(result.get("cookies", {})))
            elif status == "FREE":
                free_list.append(result)
                with open(os.path.join("hs_free", f"free_{result.get('name','unk')}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(hs_convert_to_netscape(result.get("cookies", {})))
            else:
                inv_list.append(result)

            processed += 1
            if total > 1:
                now = time.monotonic()
                if now - last_edit >= 2.0:
                    last_edit = now
                    await safe_edit(status_msg, hs_fmt_progress(
                        int(processed), total, len(live_list), len(device_list),
                        len(free_list), len(inv_list), hs_plan_breakdown(live_list),
                    ))

    user_results[uid].update({
        "hs_live": live_list, "hs_device": device_list,
        "hs_free": free_list, "hs_invalid": inv_list,
        "hs_total": total, "processing": False,
    })

    if ext != "zip" and total == 1:
        only = (live_list or device_list or free_list or inv_list)[0]
        await safe_edit(status_msg, hs_fmt_single(only), markup=main_markup("hotstar"))
        if only.get("status") != "INVALID" and GC_STORE_CHAT_ID:
            await _gc_store_hs(bot, live_list, device_list, free_list)
    else:
        await safe_edit(status_msg,
            hs_fmt_summary(total, len(live_list), len(device_list), len(free_list), len(inv_list),
                           hs_plan_breakdown(live_list)),
            markup=main_markup("hotstar"),
        )
        all_hits = [("LIVE", r) for r in live_list] + [("DEVICE_LIMIT", r) for r in device_list]
        if all_hits:
            ts_str  = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            divider = "═" * 50
            lines   = ["JioHotstar Bulk Check — Hit Report", divider,
                       f"Generated  : {ts_str}", f"Live : {len(live_list)}", f"Device Limit : {len(device_list)}",
                       divider, ""]
            for idx, (kind, res) in enumerate(all_hits, 1):
                tag = "LIVE" if kind == "LIVE" else "DEVICE LIMIT"
                lines += [f"HIT #{idx:03d} — {tag}",
                          f"  Name   : {res.get('name','?')}",
                          f"  Phone  : {res.get('phone','?')}",
                          f"  Plan   : {res.get('plan','?')}",
                          f"  Expiry : {res.get('expiry','?')}"]
                if kind == "LIVE":
                    lines += [f"  Days   : {res.get('days_left','?')} days", f"  Ads    : {res.get('show_ads','?')}"]
                lines += [f"  File   : {res.get('filename','?')}", ""]
            lines.append(divider)
            data = "\n".join(lines).encode()
            await bot.send_document(
                chat_id,
                document=BufferedInputFile(data, filename=f"hs_hits_{datetime.now():%Y%m%d_%H%M%S}.txt"),
                caption=f"📄 <b>Hotstar Hit Report</b> — {len(all_hits)} hit(s)\n"
                        f"🟢 Live: <b>{len(live_list)}</b>  ⚠️ Device Limit: <b>{len(device_list)}</b>",
                parse_mode="HTML",
            )
        if GC_STORE_CHAT_ID and (live_list or device_list or free_list):
            await _gc_store_hs(bot, live_list, device_list, free_list)


async def _run_netflix(uid, chat_id, total, files, ext, status_msg, bot: Bot):
    sub        = user_results[uid].get("nf_sub", "check")
    premium    = []; free_list = []; dead_list = []; token_list = []
    processed  = last_edit = 0.0; hit_n = 0
    loop       = asyncio.get_event_loop()
    start_t    = time.monotonic()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [loop.run_in_executor(executor, nf_check_file, fp, sub) for fp in files]
        for coro in asyncio.as_completed(futures):
            try:
                result = await coro
            except Exception as e:
                result = {"status": "DEAD", "reason": str(e)[:50], "filename": "?", "content": ""}

            status = result.get("status", "DEAD")
            if status == "PREMIUM":
                hit_n += 1; result["hit_num"] = hit_n
                premium.append(result)
                with open(os.path.join("nf_premium", f"prem_{hit_n}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(result.get("content", ""))
                await bot.send_message(chat_id, nf_fmt_premium(result), parse_mode="HTML")
            elif status == "FREE":
                free_list.append(result)
                with open(os.path.join("nf_free", f"free_{len(free_list)}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(result.get("content", ""))
            elif status == "TOKEN":
                hit_n += 1; result["hit_num"] = hit_n
                token_list.append(result)
                with open(os.path.join("nf_tokens", f"token_{hit_n}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(nf_build_token_export(result, hit_n))
                await bot.send_message(chat_id, nf_fmt_token(result), parse_mode="HTML")
            else:
                dead_list.append(result)
                with open(os.path.join("nf_dead", f"dead_{len(dead_list)}_{int(time.time())}.txt"), "w") as fh:
                    fh.write(result.get("reason", ""))

            processed += 1
            if total > 1:
                now = time.monotonic()
                if now - last_edit >= 2.0:
                    last_edit = now
                    await safe_edit(status_msg, nf_fmt_progress(
                        int(processed), total, len(premium) + len(token_list),
                        len(free_list), len(dead_list), sub,
                    ))

    elapsed = time.monotonic() - start_t
    user_results[uid].update({
        "nf_premium": premium, "nf_free": free_list,
        "nf_dead": dead_list, "nf_tokens": token_list,
        "nf_total": total, "processing": False,
    })

    if ext != "zip" and total == 1:
        only = (premium or token_list or free_list or dead_list)[0]
        if only.get("status") == "PREMIUM":
            await safe_edit(status_msg, nf_fmt_premium(only), markup=main_markup("netflix"))
        elif only.get("status") == "TOKEN":
            await safe_edit(status_msg, nf_fmt_token(only), markup=main_markup("netflix"))
        elif only.get("status") == "FREE":
            await safe_edit(status_msg, nf_fmt_free(only), markup=main_markup("netflix"))
        else:
            await safe_edit(status_msg, nf_fmt_dead(only), markup=main_markup("netflix"))
    else:
        hits = len(premium) + len(token_list)
        await safe_edit(status_msg,
            nf_fmt_summary(hits, len(free_list), len(dead_list), total, elapsed, sub),
            markup=main_markup("netflix"))
